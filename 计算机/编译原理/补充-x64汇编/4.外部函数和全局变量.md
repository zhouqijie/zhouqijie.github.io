# x64汇编的外部函数和全局变量    


### 示例：调用外部函数sqr计算全局变量pi的平方    

声明和导出：  
```asm  
;math.asm
section .data
	global pi
	pi dq 3.14159265358979323846
section .bss
section .text
	global sqr
sqr:
	push rbp
	mov rbp, rsp
	sub rsp, 8

	mulsd xmm0, xmm0
	mov rsp, rbp
	
	pop rbp
	ret
```

调用：  
```asm  
extern pi

section .data
    numf dq -3.33333
    fmtstr db "This is number: %f", 10, 0
section .text
	global main
	extern sqr
	extern printf

main:
	push rbp
	mov rbp, rsp
	
	;计算sqr  
	movsd xmm0, [rel pi]         
	movq rcx, xmm0        
	call sqr                      

	;打印  
	movsd xmm1, xmm0         
    movq rdx, xmm1                 
    mov rcx, fmtstr                
	call printf

	leave  
	ret
```
